// Esquema para extracción de datos de documentos
const documentExtractionSchema = z.object({
  documentType: z.string().describe('Tipo de documento detectado'),
  category: z.enum(['PERSONAL', 'EQUIPOS', 'LEGAL', 'AMBIENTAL']).describe('Categoría del documento'),
  expiryDate: z.string().nullable().describe('Fecha de vencimiento en formato ISO (YYYY-MM-DD) o null si no se encuentra'),
  issueDate: z.string().nullable().describe('Fecha de emisión en formato ISO (YYYY-MM-DD) o null si no se encuentra'),
  holderName: z.string().nullable().describe('Nombre del titular o responsable del documento'),
  confidence: z.number().min(0).max(1).describe('Nivel de confianza de la extracción (0-1)'),
  summary: z.string().describe('Resumen breve del contenido del documento'),
  keywords: z.array(z.string()).describe('Palabras clave extraídas del documento'),
})

export type DocumentExtraction = z.infer<typeof documentExtractionSchema>

/**
 * Extraer información de un documento usando DeepSeek
 */
export async function extractDocumentData(
  text: string
): Promise<DocumentExtraction | null> {
  if (!DEEPSEEK_API_KEY) {
    console.warn('DEEPSEEK_API_KEY no configurada')
    return null
  }

  try {
    const response = await fetch(`${DEEPSEEK_API_URL}/chat/completions`, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'Authorization': `Bearer ${DEEPSEEK_API_KEY}`,
      },
      body: JSON.stringify({
        model: 'deepseek-chat',
        messages: [
          {
            role: 'system',
            content: `Eres un experto en análisis de documentos de seguridad industrial y certificaciones HSE para Latinoamérica.
            
Tu tarea es extraer información estructurada de documentos escaneados o digitalizados.

Debes identificar:
1. Tipo de documento (certificado médico, licencia operativa, permiso ambiental, etc.)
2. Categoría (PERSONAL, EQUIPOS, LEGAL, AMBIENTAL)
3. Fecha de vencimiento (si existe)
4. Fecha de emisión (si existe)
5. Nombre del titular o responsable
6. Resumen del contenido
7. Palabras clave relevantes

Responde ÚNICAMENTE con un JSON válido con la siguiente estructura:
{
  "documentType": "string",
  "category": "PERSONAL | EQUIPOS | LEGAL | AMBIENTAL",
  "expiryDate": "YYYY-MM-DD | null",
  "issueDate": "YYYY-MM-DD | null",
  "holderName": "string | null",
  "confidence": 0.0-1.0,
  "summary": "string",
  "keywords": ["string", "string"]
}`,
          },
          {
            role: 'user',
            content: `Analiza el siguiente texto extraído de un documento y extrae la información relevante:\n\n${text}`,
          },
        ],
        temperature: 0.1,
        max_tokens: 500,
      }),
    })

    if (!response.ok) {
      const errorData = await response.json().catch(() => ({}))
      console.error('Error de DeepSeek API:', errorData)
      return null
    }

    const data = await response.json()
    const content = data.choices?.[0]?.message?.content

    if (!content) {
      return null
    }

    // Parsear la respuesta JSON
    const jsonMatch = content.match(/\{[\s\S]*\}/)
    if (!jsonMatch) {
      return null
    }

    const parsed = JSON.parse(jsonMatch[0])
    return documentExtractionSchema.parse(parsed)
  } catch (error) {
    console.error('Error extrayendo datos del documento:', error)
    return null
  }
}

/**
 * Generar mensaje de alerta personalizado
 */
export async function generateAlertMessage(params: {
  documentTitle: string
  documentType: string
  expiryDate: string
  daysRemaining: number
  holderName?: string
}): Promise<string> {
  const { documentTitle, documentType, expiryDate, daysRemaining, holderName } = params

  // Mensaje predefinido si no hay API key
  if (!DEEPSEEK_API_KEY) {
    if (daysRemaining <= 0) {
      return `⚠️ ALERTA URGENTE: ${documentType} "${documentTitle}" ${holderName ? `de ${holderName}` : ''} ha VENCIDO (${expiryDate}). Renueve inmediatamente para evitar multas.`
    }
    if (daysRemaining === 1) {
      return `⚠️ ALERTA: ${documentType} "${documentTitle}" ${holderName ? `de ${holderName}` : ''} vence MAÑANA (${expiryDate}).`
    }
    return `📅 Recordatorio: ${documentType} "${documentTitle}" ${holderName ? `de ${holderName}` : ''} vence en ${daysRemaining} días (${expiryDate}).`
  }

  try {
    const response = await fetch(`${DEEPSEEK_API_URL}/chat/completions`, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'Authorization': `Bearer ${DEEPSEEK_API_KEY}`,
      },
      body: JSON.stringify({
        model: 'deepseek-chat',
        messages: [
          {
            role: 'system',
            content: 'Eres un asistente de alertas de seguridad industrial. Genera mensajes de alerta claros y concisos en español latinoamericano.',
          },
          {
            role: 'user',
            content: `Genera un mensaje de alerta para un documento que vence:
- Tipo: ${documentType}
- Título: ${documentTitle}
- Titular: ${holderName || 'No especificado'}
- Fecha de vencimiento: ${expiryDate}
- Días restantes: ${daysRemaining}

El mensaje debe ser profesional, urgente si es necesario, y no más de 160 caracteres para WhatsApp.`,
          },
        ],
        temperature: 0.5,
        max_tokens: 100,
      }),
    })

    if (!response.ok) {
      return `📅 ${documentType} "${documentTitle}" vence en ${daysRemaining} días.`
    }

    const data = await response.json()
    return data.choices?.[0]?.message?.content || `📅 ${documentType} "${documentTitle}" vence en ${daysRemaining} días.`
  } catch (error) {
    console.error('Error generando mensaje:', error)
    return `📅 ${documentType} "${documentTitle}" vence en ${daysRemaining} días.`
  }
}

/**

 * =====================================================
 * Sistema de autenticación con NextAuth.js v4
 * Incluye roles RBAC y multi-tenancy
 */

import type { NextAuthOptions } from 'next-auth'
import CredentialsProvider from 'next-auth/providers/credentials'
import GoogleProvider from 'next-auth/providers/google'
import bcrypt from 'bcryptjs'
import { db } from '@/lib/db'

// Extender los tipos de NextAuth para incluir rol y companyId
declare module 'next-auth' {
  interface Session {
    user: {
      id: string
      email: string
      name: string | null
      role: string
      companyId: string
      companyName: string | null
    }
  }
}

declare module 'next-auth/jwt' {
  interface JWT {
    id: string
    role: string
    companyId: string
    companyName: string | null
  }
}

/**
 * Configuración de proveedores de autenticación
 * - Email/Password (Credentials)
 * - Google OAuth (opcional)
 */
export const authOptions: NextAuthOptions = {
  // NO usar adapter con JWT strategy + Credentials provider
  // El adapter es solo para database sessions
  
  // Proveedores de autenticación
  providers: [
    // Proveedor de credenciales (Email/Password)
    CredentialsProvider({
      id: 'credentials',
      name: 'Credenciales',
      credentials: {
        email: { 
          label: 'Email', 
          type: 'email', 
          placeholder: 'tu@empresa.com' 
        },
        password: { 
          label: 'Contraseña', 
          type: 'password' 
        },
      },
      async authorize(credentials) {
        console.log('[Auth] Authorize called with:', credentials?.email)
        
        // Validar que existan las credenciales
        if (!credentials?.email || !credentials?.password) {
          console.log('[Auth] Missing credentials')
          return null
        }

        try {
          // Buscar usuario por email
          const user = await db.user.findUnique({
            where: { email: credentials.email.toLowerCase() },
            include: { company: true },
          })

          // Si no existe el usuario, retornar null
          if (!user) {
            console.log('[Auth] User not found:', credentials.email)
            return null
          }

          // Si no tiene passwordHash, no puede hacer login con credenciales
          if (!user.passwordHash) {
            console.log('[Auth] User has no password hash')
            return null
          }

          // Verificar que el usuario está activo
          if (user.isActive === false) {
            console.log('[Auth] User is inactive')
            return null
          }

          // Verificar contraseña
          console.log('[Auth] Comparing passwords...')
          const isValidPassword = await bcrypt.compare(
            credentials.password,
            user.passwordHash
          )

          if (!isValidPassword) {
            console.log('[Auth] Invalid password for user:', credentials.email)
            return null
          }

          console.log('[Auth] Login successful for:', user.email)

          // Actualizar último login (sin esperar)
          db.user.update({
            where: { id: user.id },
            data: { lastLoginAt: new Date() },
          }).catch(console.error)

          // Registrar en audit log (sin esperar)
          db.auditLog.create({
            data: {
              companyId: user.companyId,
              userId: user.id,
              action: 'LOGIN',
              entityType: 'USER',
              entityId: user.id,
              details: JSON.stringify({ method: 'credentials' }),
            },
          }).catch(console.error)

          // Retornar el usuario con los campos necesarios
          return {
            id: user.id,
            email: user.email,
            name: user.name,
            role: user.role,
            companyId: user.companyId,
            companyName: user.company?.name || null,
          }
        } catch (error) {
          console.error('Error en authorize:', error)
          return null
        }
      },
    }),

    // Proveedor de Google OAuth (configurar si se desea usar)
    GoogleProvider({
      clientId: process.env.GOOGLE_CLIENT_ID || '',
      clientSecret: process.env.GOOGLE_CLIENT_SECRET || '',
    }),
  ],

  // Configuración de sesiones
  session: {
    strategy: 'jwt', // Usar JWT para mejor performance
    maxAge: 30 * 24 * 60 * 60, // 30 días
  },

  // Páginas personalizadas
  pages: {
    signIn: '/', // Página de login (dashboard principal)
    error: '/', // Página de error
  },

  // Callbacks para personalizar el token y la sesión
  callbacks: {
    // Al crear el JWT, incluir datos adicionales
    async jwt({ token, user, trigger, session }) {
      // Si hay usuario nuevo (login inicial), agregar datos
      if (user) {
        token.id = user.id || ''
        token.role = (user as any).role || 'VIEWER'
        token.companyId = (user as any).companyId || ''
        token.companyName = (user as any).companyName || null
      }

      // Si se actualiza la sesión, refrescar datos del usuario
      if (trigger === 'update' && session) {
        const dbUser = await db.user.findUnique({
          where: { id: token.id },
          include: { company: true },
        })
        if (dbUser) {
          token.role = dbUser.role
          token.companyId = dbUser.companyId
          token.companyName = dbUser.company?.name || null
        }
      }

      return token
    },

    // Al crear la sesión, pasar datos del JWT
    async session({ session, token }) {
      if (token && session.user) {
        session.user.id = token.id
        session.user.role = token.role
        session.user.companyId = token.companyId
        session.user.companyName = token.companyName
      }
      return session
    },

    // Al hacer signIn con OAuth
    async signIn({ user, account }) {
      // Para credenciales, ya se validó en authorize()
      if (account?.provider === 'credentials') {
        return true
      }

      // Para OAuth (Google), crear usuario si no existe
      if (account?.provider === 'google' && user.email) {
        const existingUser = await db.user.findUnique({
          where: { email: user.email },
        })

        if (!existingUser) {
          // Crear empresa y usuario nuevos
          const newCompany = await db.company.create({
            data: {
              name: `Empresa de ${user.name || 'Nuevo Usuario'}`,
              subscriptionTier: 'FREE',
              subscriptionStatus: 'TRIAL',
            },
          })

          await db.user.create({
            data: {
              email: user.email,
              name: user.name,
              role: 'ADMIN',
              companyId: newCompany.id,
            },
          })
        }

        return true
      }

      return true
    },
  },

  // Eventos para logging
  events: {
    async signOut({ token }) {
      // Registrar logout en audit log
      if (token?.id) {
        const user = await db.user.findUnique({ where: { id: token.id } })
        if (user) {
          await db.auditLog.create({
            data: {
              companyId: user.companyId,
              userId: user.id,
              action: 'LOGOUT',
              entityType: 'USER',
              entityId: user.id,
            },
          }).catch(console.error)
        }
      }
    },
  },

  // Habilitar debug en desarrollo
  debug: process.env.NODE_ENV === 'development',

  // Secret para encriptar tokens
  secret: process.env.NEXTAUTH_SECRET,

  // Configuración de cookies para entornos con proxy/gateway
  // NOTA: Para producción, usar sameSite: 'lax' y secure: true
  cookies: {
    sessionToken: {
      name: 'next-auth.session-token',
      options: {
        httpOnly: true,
        sameSite: 'lax',
        path: '/',
        secure: false,
      },
    },
  },
}

/**
 * Utilidades para verificar roles
 */
export const ROLES = {
  ADMIN: 'ADMIN',
  MANAGER: 'MANAGER',
  VIEWER: 'VIEWER',
} as const

export type Role = typeof ROLES[keyof typeof ROLES]

/**
 * Verificar si un usuario tiene al menos el rol especificado
 * ADMIN > MANAGER > VIEWER
 */
export function hasRequiredRole(userRole: string, requiredRole: Role): boolean {
  const roleHierarchy = {
    ADMIN: 3,
    MANAGER: 2,
    VIEWER: 1,
  }

  const userLevel = roleHierarchy[userRole as Role] || 0
  const requiredLevel = roleHierarchy[requiredRole] || 0

  return userLevel >= requiredLevel
}

/**
 * Verificar si el usuario puede gestionar documentos
 */
export function canManageDocuments(role: string): boolean {
  return role === ROLES.ADMIN || role === ROLES.MANAGER
}

/**
 * Verificar si el usuario puede ver todos los documentos
 */
export function canViewAllDocuments(role: string): boolean {
  return role === ROLES.ADMIN || role === ROLES.MANAGER
}

/**
 * Verificar si el usuario puede gestionar usuarios
 */
export function canManageUsers(role: string): boolean {
  return role === ROLES.ADMIN
}

/**
 * Verificar si el usuario puede gestionar la suscripción
 */
export function canManageSubscription(role: string): boolean {
  return role === ROLES.ADMIN
}
